```
const onChangeFilter = (state) => {
    setFilterStates(state);

    let filterParams = new URLSearchParams();                                   //  쿼리 스트링 저장하는 변수

    if (query.get("search")) filterParams.set("search",query.get("search"));    //  필터에 영향을 받지 않는 스트링 탐색
    if (query.get("cate1")) filterParams.set("cate1", query.get("cate1"));
    if (query.get("cate2")) filterParams.set("cate2", query.get("cate2"));
    filterParams.set('filter','True')

    Object.keys(state).forEach( key => {
      if(Array.isArray(state[key])) 
        state[key].forEach( value => filterParams.append(key, value))
      else 
        filterParams.append(key, state[key])
    })
    
    const newURL = `${pathname}?${filterParams}`;
    navigate(newURL);
  }
```
